    Interface IArchetypeProps<Id, Dimension, Physical>

    What a producer built: the archetype at rest, which Archetype reads as Archetype.IProps.

    interface IArchetypeProps<
        Id extends string,
        Dimension extends string,
        Physical extends IProperties | null = IProperties,
    > {
        affordances: (IAutoMovieAffordance & { id: Id })[];
        archetype: string;
        cells?: Table;
        contact: IAutoMovieBodyContact | null;
        dimensions: Readonly<Partial<Record<Dimension, number>>>;
        drag: IAutoMovieBodyDrag | null;
        fractures: readonly ISet[];
        friction: number | null;
        id: string;
        kineticFriction: number | null;
        masses: Readonly<Record<string, number>>;
        material: ArchetypeMaterial.Kind | null;
        materials: IAutoMovieMaterial[];
        name: string;
        parts: IAutoMovieModelPart[];
        properties: Physical;
        provenance: IProvenance<Dimension>;
        restitution: number | null;
        spans: readonly ArchetypeSpan.ISpan[];
        surfaces: Readonly<Record<string, Material>>;
    }

    Type Parameters

    • Id extends string

      The archetype's affordance ids, a closed union the producer declares.

    • Dimension extends string

      The archetype's free dimensions, a closed union the producer declares.

    • Physical extends IProperties | null = IProperties

      Mass properties of a solid or null for a visual-only archetype.
