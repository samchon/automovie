    Interface IViewerCameraStateSnapshot

    Detached native camera state admitted by createViewerCameraState.restore. Vectors and lens values are local native camera data, not a world hierarchy.

    interface IViewerCameraStateSnapshot {
        layers: number;
        lens:
            | {
                aspect: number;
                far: number;
                filmGauge: number;
                filmOffset: number;
                focus: number;
                fov: number;
                kind: "perspective";
                near: number;
                zoom: number;
            }
            | {
                bottom: number;
                far: number;
                kind: "orthographic";
                left: number;
                near: number;
                right: number;
                top: number;
                zoom: number;
            };
        position: Vector3Tuple;
        projection: Matrix4Tuple;
        quaternion: QuaternionTuple;
        scale: Vector3Tuple;
        type: string;
        up: Vector3Tuple;
        view: | {
            enabled: boolean;
            fullHeight: number;
            fullWidth: number;
            height: number;
            offsetX: number;
            offsetY: number;
            width: number;
        }
        | null;
    }
